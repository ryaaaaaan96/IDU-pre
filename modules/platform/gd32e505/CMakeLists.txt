# 修复上游 APB 分频字段编码判断，所有标准 PWM/counter 使用相同正确时钟。
if(CONFIG_SOC_GD32E505 AND CONFIG_CLOCK_CONTROL_GD32)
  if(NOT TARGET drivers__clock_control)
    message(FATAL_ERROR "GD32 clock target is missing")
  endif()
  include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/clock_control.cmake)
  set(patched_clock ${CMAKE_CURRENT_BINARY_DIR}/clock_control_gd32.c)
  gd32e505_patch_clock(
    ${ZEPHYR_BASE}/drivers/clock_control/clock_control_gd32.c ${patched_clock})
  get_target_property(clock_sources drivers__clock_control SOURCES)
  set(upstream_clock_sources ${clock_sources})
  list(FILTER upstream_clock_sources INCLUDE REGEX "(^|/)clock_control_gd32\\.c$")
  list(LENGTH upstream_clock_sources upstream_clock_count)
  if(NOT upstream_clock_count EQUAL 1)
    message(FATAL_ERROR "GD32 clock override: expected exactly one upstream source")
  endif()
  list(FILTER clock_sources EXCLUDE REGEX "(^|/)clock_control_gd32\\.c$")
  list(APPEND clock_sources ${patched_clock})
  set_property(TARGET drivers__clock_control PROPERTY SOURCES ${clock_sources})
endif()

# 修正上游 GD32 PWM 的 CAR 边界，使请求周期等于实际周期计数。
if(CONFIG_SOC_GD32E505 AND CONFIG_PWM_GD32)
  if(NOT TARGET drivers__pwm)
    message(FATAL_ERROR "GD32 PWM target is missing")
  endif()
  include(${CMAKE_CURRENT_SOURCE_DIR}/cmake/pwm_gd32.cmake)
  set(patched_pwm ${CMAKE_CURRENT_BINARY_DIR}/pwm_gd32.c)
  gd32e505_patch_pwm(${ZEPHYR_BASE}/drivers/pwm/pwm_gd32.c ${patched_pwm})
  get_target_property(pwm_sources drivers__pwm SOURCES)
  set(upstream_pwm_sources ${pwm_sources})
  list(FILTER upstream_pwm_sources INCLUDE REGEX "(^|/)pwm_gd32\\.c$")
  list(LENGTH upstream_pwm_sources upstream_pwm_count)
  if(NOT upstream_pwm_count EQUAL 1)
    message(FATAL_ERROR "GD32 PWM override: expected exactly one upstream source")
  endif()
  list(FILTER pwm_sources EXCLUDE REGEX "(^|/)pwm_gd32\\.c$")
  list(APPEND pwm_sources ${patched_pwm})
  set_property(TARGET drivers__pwm PROPERTY SOURCES ${pwm_sources})
endif()

# 能力声明不能静默落回没有 async ops 的上游驱动。
if(CONFIG_SOC_GD32E505 AND CONFIG_USART_GD32 AND CONFIG_UART_ASYNC_API
   AND NOT CONFIG_UART_GD32_ASYNC)
  message(FATAL_ERROR "GD32 UART async requires CONFIG_UART_GD32_ASYNC=y and DMA_GD32")
endif()

if(CONFIG_UART_GD32_ASYNC)
  if(NOT TARGET drivers__serial)
    message(FATAL_ERROR "GD32 async override: upstream serial target is missing")
  endif()
  set(upstream_uart ${ZEPHYR_BASE}/drivers/serial/usart_gd32.c)
  file(SHA256 ${upstream_uart} upstream_uart_sha256)
  # Zephyr 升级时须人工合并上游差异，再更新此基线，禁止直接绕过检查。
  if(NOT upstream_uart_sha256 STREQUAL
     "9054bd41a93f91ef19d24ffffb63d429a07833d5739b99e5457d9c495efd613c")
    message(FATAL_ERROR "GD32 async override: upstream UART changed; review and merge before updating hash")
  endif()
  get_target_property(serial_sources drivers__serial SOURCES)
  set(upstream_sources ${serial_sources})
  list(FILTER upstream_sources INCLUDE REGEX "(^|/)usart_gd32\\.c$")
  list(LENGTH upstream_sources upstream_count)
  if(NOT upstream_count EQUAL 1)
    message(FATAL_ERROR "GD32 async override: expected exactly one upstream UART source")
  endif()
  list(FILTER serial_sources EXCLUDE REGEX "(^|/)usart_gd32\\.c$")
  list(APPEND serial_sources ${CMAKE_CURRENT_SOURCE_DIR}/drivers/serial/usart_gd32.c)
  set_property(TARGET drivers__serial PROPERTY SOURCES ${serial_sources})
endif()
